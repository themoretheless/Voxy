cmake_minimum_required(VERSION 3.20)
project(voxy_streamline LANGUAGES CXX)
set(VOXY_STREAMLINE_SDK "" CACHE PATH "Official Streamline SDK root (2.14.1)")
if(NOT EXISTS "${VOXY_STREAMLINE_SDK}/include/sl_dlss_g.h")
    message(FATAL_ERROR "Set VOXY_STREAMLINE_SDK to the official SDK root")
endif()
add_library(voxy_streamline_fg STATIC frame_generation.cpp reflex.cpp frame_token.cpp session.cpp dlss.cpp)
target_compile_features(voxy_streamline_fg PUBLIC cxx_std_17)
target_include_directories(voxy_streamline_fg PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(voxy_streamline_fg SYSTEM PUBLIC "${VOXY_STREAMLINE_SDK}/include")
include(CTest)
if(BUILD_TESTING)
    add_executable(voxy_streamline_fg_test frame_generation_test.cpp)
    target_link_libraries(voxy_streamline_fg_test PRIVATE voxy_streamline_fg)
    # Tests use assertions even with a release build.
    if(MSVC)
        target_compile_options(voxy_streamline_fg_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(voxy_streamline_fg_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME frame_generation_policy COMMAND voxy_streamline_fg_test)
endif()

option(VOXY_STREAMLINE_VULKAN "Enable Vulkan device registration" OFF)
set(VOXY_VULKAN_HEADERS "" CACHE PATH "Khronos Vulkan-Headers include directory")
if(VOXY_STREAMLINE_VULKAN)
    if(NOT EXISTS "${VOXY_VULKAN_HEADERS}/vulkan/vulkan.h")
        message(FATAL_ERROR "Set VOXY_VULKAN_HEADERS to official Vulkan-Headers/include")
    endif()
    target_compile_definitions(voxy_streamline_fg PUBLIC VOXY_STREAMLINE_VULKAN)
    target_include_directories(voxy_streamline_fg SYSTEM PUBLIC "${VOXY_VULKAN_HEADERS}")
endif()

if(WIN32)
    target_sources(voxy_streamline_fg PRIVATE windows_module.cpp windows_runtime.cpp c_api.cpp)
endif()
